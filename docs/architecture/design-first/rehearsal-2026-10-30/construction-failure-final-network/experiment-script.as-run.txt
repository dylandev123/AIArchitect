/** Authorized two-call experiment; architectural review must release stage two. */
import assert from 'node:assert/strict';
import { readFileSync, writeFileSync, mkdirSync, existsSync, readdirSync } from 'node:fs';
import { createHash } from 'node:crypto';
import { loadEnvConfig } from '@next/env';
import { Pool, Client } from 'pg';
import { withDesignCheckpoint, designProposalHash, type DesignCheckpointReview } from '../src/lib/villa/scene/designCheckpoint';
import { proposalFixture } from '../src/lib/villa/scene/designProposal.fixture';
import { mansionDesignCriteria } from '../src/lib/villa/scene/designCheckpoint';
import { getAiModelId } from '../src/lib/ai/model';
import { sceneRequestSchema } from '../src/lib/villa/scene/terra';
import { getPricingTable, computeCostUsd } from '../src/lib/ai/pricing';

const rehearsal = process.argv.find(a => a.startsWith('--rehearsal='))?.split('=')[1];
assert(!rehearsal || ['pass', 'reject', 'provider-failure', 'construction-failure'].includes(rehearsal));
assert(rehearsal || process.argv.includes('--execute-authorized-experiment'), 'Explicit execution flag required.');
const dir = rehearsal ? `docs/architecture/design-first/rehearsal-2026-10-30/${rehearsal}${process.argv.includes("--final-check") ? "-final" : ""}${process.argv.includes("--network-enabled") ? "-network" : ""}` : 'docs/architecture/design-first/live-after-rehearsal-2026-10-30';
assert(!existsSync(dir), 'Experiment already started; never resubmit.');
loadEnvConfig(process.cwd());
assert(process.env.OPENAI_API_KEY?.trim(), 'Existing API configuration unavailable.');
assert.equal(getAiModelId(), 'gpt-5.6-terra', 'Model changed; stop.');
const input = sceneRequestSchema.parse(JSON.parse(readFileSync('docs/architecture/design-first/experiment.request.json', 'utf8')));
assert.equal(input.mode, 'generate');
assert(!input.testing && !input.current && !input.reference, 'Use original independent mansion brief only.');
mkdirSync(dir, { recursive: true, mode: 0o700 });
const save = (name: string, value: unknown) => writeFileSync(`${dir}/${name}`, typeof value === 'string' ? value : JSON.stringify(value, null, 2), { flag: 'wx', mode: 0o600 });
const hash = (text: string | Buffer) => createHash('sha256').update(text).digest('hex');
function files(root: string): string[] { return readdirSync(root, { withFileTypes: true }).flatMap(e => e.isDirectory() ? files(`${root}/${e.name}`) : [`${root}/${e.name}`]); }
const historicalRoots = ['docs/architecture/complete-mansion-2026-10-30', 'docs/architecture/generation-quality', 'docs/architecture/terra-estate-diagnosis'];
const historicalHashes = Object.fromEntries(historicalRoots.flatMap(files).map(path => [path, hash(readFileSync(path))]));
save('historical-hashes.before.json', historicalHashes);
save('source-hashes.json', Object.fromEntries([...files('src'), 'scripts/design-first-experiment.ts', 'package-lock.json'].map(path => [path, hash(readFileSync(path))])));
save('request.json', input);
save('authorization.json', { startedAt: new Date().toISOString(), rehearsal: rehearsal ?? null, userAuthorization: rehearsal ? 'OFFLINE ONLY; mock provider; synthetic checkpoint review' : 'ONE controlled design-first mansion experiment; max two GPT-5.6 Terra calls; construction conditional on seven-criterion pass.', model: getAiModelId(), maxPaidCalls: rehearsal ? 0 : 2, maxRetries: 0, reasoning: 'provider-default', maxOutputTokensPerStage: 30000, execution: 'Local direct invocation of existing route; no hosting time limit or HTTP proxy. Existing per-stage abort limits remain.', databasePolicy: 'SELECT only; server read-only sessions; usage intercepted locally.' });
const dbEvents: unknown[] = [];
const originals = new Map<any, any>();
let usageRecords = 0;
let environmentFailed = false;
process.env.PGOPTIONS = '-c default_transaction_read_only=on';
process.env.AI_USAGE_LOG_PATH = `${process.cwd()}/${dir}/usage.jsonl`;
for (const prototype of [Pool.prototype, Client.prototype]) {
    const original = prototype.query;
    originals.set(prototype, original);
    (prototype as any).query = function (...args: any[]) {
        const sql = typeof args[0] === 'string' ? args[0] : args[0]?.text;
        assert.equal(typeof sql, 'string');
        const callbackIndex = typeof args.at(-1) === 'function' ? args.length - 1 : -1;
        const complete = (result: unknown) => {
            if (callbackIndex < 0) return Promise.resolve(result);
            queueMicrotask(() => args[callbackIndex](null, result));
            return undefined;
        };
        if (/^\s*CREATE TABLE/i.test(sql)) { dbEvents.push({ suppressed: 'initializer DDL', sha256: hash(sql) }); return complete({ rows: [], rowCount: 0 }); }
        if (/^\s*INSERT INTO ai_usage\b/i.test(sql)) { save(`usage-record-${++usageRecords}.json`, args[1]); dbEvents.push({ redirected: 'usage to experiment evidence' }); return complete({ rows: [], rowCount: 0 }); }
        assert(/^\s*SELECT\b/i.test(sql) && !sql.includes(';'), 'Database mutation or multi-statement blocked.');
        assert(!environmentFailed, 'Execution already failed; no further database reads.');
        dbEvents.push({ read: sql });
        if (callbackIndex >= 0) {
            const callback = args[callbackIndex];
            args[callbackIndex] = (error: Error | null, result: any) => {
                if (error) environmentFailed = true;
                else dbEvents.push({ completed: 'callback', rows: result.rowCount });
                callback(error, result);
            };
            return (original as any).apply(this, args);
        }
        return (original as any).apply(this, args).then((result: any) => {
            dbEvents.push({ completed: 'promise', rows: result.rowCount });
            return result;
        }, (error: Error) => { environmentFailed = true; throw error; });
    };
}
let paidCalls = 0;
let checkpointPassed = false;
const nativeFetch = globalThis.fetch;
globalThis.fetch = async (url, init) => {
    assert(!environmentFailed, 'Prior execution failure; no provider request allowed.');
    assert.equal(String(url instanceof Request ? url.url : url), 'https://api.openai.com/v1/responses');
    if (paidCalls === 0) {
        const pool = (globalThis as any).__libraryPool?.pool;
        assert(pool, 'Real database retrieval required before provider submission.');
        const result = await pool.query("SELECT current_setting('transaction_read_only') AS read_only");
        assert.equal(result.rows[0].read_only, 'on');
        save('database-read-only.json', { transactionReadOnly: result.rows[0].read_only });
    }
    const body = String(init?.body), wire = JSON.parse(body);
    assert.equal(wire.model, 'gpt-5.6-terra');
    assert.equal(wire.max_output_tokens, 30000);
    assert.equal(wire.text.format.strict, true);
    assert.equal(wire.reasoning?.effort, undefined);
    assert(paidCalls < 2, 'Third provider request forbidden.');
    if (paidCalls === 1) assert(checkpointPassed, 'Construction requires an approved exact-hash checkpoint.');
    const stage = paidCalls === 0 ? 'design' : 'construction';
    assert.equal(Object.hasOwn(wire.text.format.schema.properties, 'operations'), stage === 'construction');
    save(`${stage}.provider-request.exact.json`, body);
    save(`${stage}.call-started.json`, { mocked: !!rehearsal, at: new Date().toISOString(), requestHash: hash(body), ordinal: ++paidCalls });
    console.log(JSON.stringify({ stage, providerRequestStarted: paidCalls, mocked: !!rehearsal }));
    try {
        let response: Response;
        if (rehearsal) {
            const fail = rehearsal === 'provider-failure' || (rehearsal === 'construction-failure' && stage === 'construction');
            const fixtureRoot = 'docs/architecture/complete-mansion-2026-10-30';
            const scene = JSON.parse(readFileSync(`${fixtureRoot}/scene.json`, 'utf8'));
            const value = stage === 'design' ? proposalFixture(scene, ['main-living', 'suite-west', 'suite-east']) : JSON.parse(readFileSync(`${fixtureRoot}/normalized-output.json`, 'utf8'));
            response = new Response(JSON.stringify(fail ? { error: { message: 'Deliberate offline provider failure', type: 'server_error' } } : { id: `resp_mock_${stage}`, object: 'response', created_at: 0, model: wire.model, status: 'completed', output: [{ type: 'message', id: `msg_${stage}`, role: 'assistant', content: [{ type: 'output_text', text: JSON.stringify(value), annotations: [] }] }], usage: { input_tokens: 1, output_tokens: 1, total_tokens: 2 } }), { status: fail ? 500 : 200, headers: { 'content-type': 'application/json' } });
        } else response = await nativeFetch(url, { ...init, redirect: 'error' });
        const exact = await response.clone().text();
        save(`${stage}.provider-response.exact.json`, exact);
        save(`${stage}.provider-http.json`, { status: response.status, requestId: response.headers.get('x-request-id'), endedAt: new Date().toISOString() });
        if (!response.ok) environmentFailed = true;
        return response;
    } catch (error) {
        environmentFailed = true;
        save(`${stage}.transport-error.json`, { type: error instanceof Error ? error.name : 'unknown', note: 'Stop; no retry. Provider may have received the request.' });
        throw error;
    }
};
try {
    save('experiment-script.as-run.txt', readFileSync('scripts/design-first-experiment.ts', 'utf8'));
    const { POST } = await import('../src/app/api/ai/scene/route');
    const response = await withDesignCheckpoint(async proposal => {
        assert(!environmentFailed);
        const proposalHash = designProposalHash(proposal);
        save('proposal.json', proposal);
        save('checkpoint.pending.json', { proposalHash, status: 'AWAITING_ARCHITECTURAL_REVIEW', paidCalls, constructionAllowed: false });
        console.log(JSON.stringify({ checkpoint: 'WAITING', proposalHash, paidCalls }));
        if (rehearsal) save('checkpoint.review.json', {
            proposalHash, criteria: Object.fromEntries(mansionDesignCriteria.map(key => [key, { verdict: rehearsal === 'reject' ? 'fail' : 'pass', evidence: 'MOCK REVIEW ONLY: exercises seven-criterion gate; no architectural merit certified.' }])),
            conclusion: 'Synthetic rehearsal decision only; never authorize a live proposal with this review.',
        });
        // Keep this same route invocation alive. No proposal regeneration/replay.
        const deadline = Date.now() + 30 * 60 * 1000;
        while (!existsSync(`${dir}/checkpoint.review.json`)) {
            assert(Date.now() < deadline, 'Checkpoint review deadline exceeded; stop.');
            await new Promise(resolve => setTimeout(resolve, 1000));
        }
        const review = JSON.parse(readFileSync(`${dir}/checkpoint.review.json`, 'utf8')) as DesignCheckpointReview;
        // Existing checkpoint performs the authoritative seven-criterion gate.
        assert.equal(review.proposalHash, proposalHash);
        checkpointPassed = Object.values(review.criteria).length === 7 && Object.values(review.criteria).every(c => c.verdict === 'pass' && c.evidence.trim().length > 0) && !!review.conclusion.trim();
        return review;
    }, () => POST(new Request('http://localhost/api/ai/scene', { method: 'POST', body: JSON.stringify(input) })));
    const body = await response.json();
    save('route-response.json', body);
    if (body.responseId) for (const suffix of ['.json', '.wire.json', '.diagnostics.json', '.provider.txt']) {
        const path = `.data/scene-responses/${body.responseId}${suffix}`;
        if (existsSync(path)) save(`retained${suffix}`, readFileSync(path, 'utf8'));
    }
    if (body.scene) save('scene.json', body.scene);
    if (body.recoverableDraft) save('draft.REJECTED.json', body.recoverableDraft);
    if (rehearsal) {
        assert(dbEvents.some((e: any) => e.completed === 'callback'), 'Real Pool-to-Client callback was not exercised');
        assert(dbEvents.some((e: any) => e.completed === 'promise'), 'Real Pool Promise did not complete');
        assert.equal(paidCalls, ['pass', 'construction-failure'].includes(rehearsal) ? 2 : 1);
        assert.equal(!!body.scene, rehearsal === 'pass', JSON.stringify(body));
        if (rehearsal === 'reject') assert.match(body.error, /architectural checkpoint/);
        save('rehearsal-assertions.json', { passed: true, providerMockOnly: true, paidCalls: 0, mockProviderRequests: paidCalls, accepted: !!body.scene, editableOperations: body.operations?.length, fidelity: body.designFidelity?.status });
    }
    save('result.json', { status: response.status, paidCalls: rehearsal ? 0 : paidCalls, mockProviderRequests: rehearsal ? paidCalls : 0, checkpointPassed, environmentFailed, responseId: body.responseId, error: body.error, dbEvents });
    console.log(JSON.stringify({ finished: true, status: response.status, paidCalls, checkpointPassed, accepted: !!body.scene, error: body.error }));
} catch (error) {
    save('execution-error.json', { type: error instanceof Error ? error.name : 'unknown', message: String(error), paidCalls, dbEvents });
    console.log(JSON.stringify({ stopped: true, paidCalls, error: String(error) }));
} finally {
    globalThis.fetch = nativeFetch;
    for (const [prototype, original] of originals) prototype.query = original;
    const pools = globalThis as any;
    for (const key of ['__libraryPool', '__aiUsagePool']) if (pools[key]) { await pools[key].pool.end(); delete pools[key]; }
    save('cleanup.json', { fetchRestored: globalThis.fetch === nativeFetch, queryPrototypesRestored: [...originals].every(([p, q]) => p.query === q), poolsClosed: true });
    const stages = ['design', 'construction'].flatMap(stage => {
        const path = `${dir}/${stage}.provider-response.exact.json`;
        if (!existsSync(path)) return [];
        try {
            const wire = JSON.parse(readFileSync(path, 'utf8')), u = wire.usage;
            const rates = getPricingTable()['gpt-5.6-terra'];
            const cost = u ? computeCostUsd('gpt-5.6-terra', { inputTokens: u.input_tokens, cachedInputTokens: u.input_tokens_details?.cached_tokens ?? 0, outputTokens: u.output_tokens }) : null;
            return [{ stage, providerId: wire.id, model: wire.model, status: wire.status, usage: u ?? null, ratesUsdPerMillion: rates, estimatedCostUsd: cost }];
        } catch { return [{ stage, providerId: null, model: null, status: null, usage: null, ratesUsdPerMillion: null, estimatedCostUsd: null }]; }
    });
    save('usage-and-cost.json', { paidCalls: rehearsal ? 0 : paidCalls, mockProviderRequests: rehearsal ? paidCalls : 0, stages, estimatedTotalUsd: rehearsal ? 0 : stages.every(s => s.estimatedCostUsd !== null) ? stages.reduce((sum, s) => sum + s.estimatedCostUsd!, 0) : null, source: 'Repository configured pricing, estimate only; provider token counts. Cache-write premium not included; not invoice reconciliation.' });
    const changed = Object.entries(historicalHashes).filter(([path, digest]) => hash(readFileSync(path)) !== digest).map(([path]) => path);
    save('preservation.json', { checkedFiles: Object.keys(historicalHashes).length, changed, previousEvidenceUnchanged: changed.length === 0 });
}
process.exit(rehearsal && !existsSync(`${dir}/rehearsal-assertions.json`) ? 1 : 0);
